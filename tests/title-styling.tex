\documentclass[CLASS_OPTIONS]{DOCUMENT_CLASS}
\usepackage[title,headings=false,code=false,boxes=false,macros=false]{technica}
\usepackage[pdfusetitle]{hyperref}
\makeatletter
\newcommand{\checktitle}{%
  \ifx\f@family\sfdefault\else\errmessage{Wrong title family}\fi
  \begingroup\bfseries\xdef\expectedseries{\f@series}\endgroup
  \ifx\f@series\expectedseries\else\errmessage{Wrong title weight}\fi
  \let\actualsize\f@size
  \begingroup\LARGE\xdef\expectedsize{\f@size}\endgroup
  \ifx\actualsize\expectedsize\else\errmessage{Title size changed}\fi
  \let\actualink\current@color
  \begingroup\color{TechnicaInk}\global\let\expectedink\current@color\endgroup
  \ifx\actualink\expectedink\else\errmessage{Wrong title color}\fi
}
\newcommand{\checkbodyfont}{%
  \ifx\f@family\rmdefault\else\errmessage{Title family leaked}\fi
  \begingroup\normalfont\xdef\expectedseries{\f@series}\endgroup
  \ifx\f@series\expectedseries\else\errmessage{Title weight leaked}\fi
}
\newcommand{\checkmetadatafont}{%
  \ifx\f@family\sfdefault\else\errmessage{Wrong author or date family}\fi
  \begingroup\normalfont\xdef\expectedseries{\f@series}\endgroup
  \ifx\f@series\expectedseries\else\errmessage{Wrong author or date weight}\fi
}
\pdfstringdefDisableCommands{\let\checktitle\@empty\let\checkbodyfont\@empty\let\checkmetadatafont\@empty}
\makeatother
% A declaration after begin-document must also receive the title styling.
\begin{document}
\title{\checktitle Styled title\thanks{Acknowledgment retained.}}
\author{\checkmetadatafont Author retained\thanks{Author note retained.}\and\checkmetadatafont Second author\\\checkmetadatafont Affiliation}
\date{\checkmetadatafont Date retained}
\maketitle
\checkbodyfont Body retained.
\end{document}
