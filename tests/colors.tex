\newcommand{\assertcolor}[2]{%
  \begingroup
  \convertcolorspec{named}{#1}{HTML}\actualcolor
  \edef\expectedcolor{#2}%
  \edef\recordedcolor{\csname #1Hex\endcsname}%
  \ifx\actualcolor\expectedcolor\else
    \errmessage{Wrong color for #1: \actualcolor, expected #2}%
  \fi
  \ifx\recordedcolor\expectedcolor\else
    \errmessage{Hex value disagrees with color #1}%
  \fi
  \endgroup
}

% Reject cycles without disturbing the existing bindings or values.
\newcommand{\assertaliascycle}[2]{%
  \begingroup
  \def\goterror{no}%
  \def\PackageError##1##2##3{\def\goterror{yes}}%
  \technicaaliascolor{#1}{#2}%
  \def\expectederror{yes}%
  \ifx\goterror\expectederror\else\errmessage{Alias cycle accepted}\fi
  \assertcolor{Leaf}{AABBCC}%
  \endgroup
}
\assertaliascycle{Dependent}{Dependent}
\assertaliascycle{TechnicaGlaze}{Leaf}
\technicadefinecolor{TechnicaGlaze}{123456}
\assertcolor{Leaf}{123456}
\technicaaliascolor{ExternalRed}{red}
\assertcolor{ExternalRed}{FF0000}
